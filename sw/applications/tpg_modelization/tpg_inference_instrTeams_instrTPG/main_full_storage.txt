#include <stdio.h>
#include <stdlib.h>
#include <stdint.h>
#include <math.h>

#include "csr.h"
#include "codegen/externHeader.h"
#include "precalcul/LE_states.h"
#include "codegen/TPG.h"

/* Print config */
#define PRINTF_IN_FPGA  1
#define PRINTF_IN_SIM   1

#if TARGET_SIM && PRINTF_IN_SIM
    #define PRINTF(fmt, ...)    printf(fmt, ## __VA_ARGS__)
#elif PRINTF_IN_FPGA
    #define PRINTF(fmt, ...)    printf(fmt, ## __VA_ARGS__)
#else
    #define PRINTF(...)
#endif

/* Globals */
typeInf* in1;
typeInf* in2;
typeInf* in3;
typeInf* in4;

/* Buffers */
typeInf in1_buf[3];
typeInf in2_buf[3];
typeInf in3_buf[3];
typeInf in4_buf[6];

/* Assign inputs */
static void assign_LE_values(int seed)
{
    in1_buf[0] = dataSourcesLE_0[seed];
    in1_buf[1] = dataSourcesLE_1[seed];
    in1_buf[2] = dataSourcesLE_2[seed];

    in2_buf[0] = dataSourcesLE_3[0];
    in2_buf[1] = dataSourcesLE_4[0];
    in2_buf[2] = dataSourcesLE_5[0];

    in3_buf[0] = dataSourcesLE_6[seed];
    in3_buf[1] = dataSourcesLE_7[seed];
    in3_buf[2] = dataSourcesLE_8[seed];

    in4_buf[0] = dataSourcesLE_9[0];
    in4_buf[1] = dataSourcesLE_10[0];
    in4_buf[2] = dataSourcesLE_11[0];
    in4_buf[3] = dataSourcesLE_12[0];
    in4_buf[4] = dataSourcesLE_13[0];
    in4_buf[5] = dataSourcesLE_14[0];

    in1 = in1_buf;
    in2 = in2_buf;
    in3 = in3_buf;
    in4 = in4_buf;
}

/* ========================= */
/* ===== RAW STORAGE ======= */
/* ========================= */

typedef struct {
    uint32_t values[NB_VALUES_PER_CLASS];
    uint32_t count; // Number of valid samples stored
} ClassSamples;

int main(void)
{
    CSR_CLEAR_BITS(CSR_REG_MCOUNTINHIBIT, 0x1);

    /* Per-class storage */
    ClassSamples samples[NB_CLASSES];
    
    // Team cycle_counters
    uint32_t team_cycles[NB_TEAMS];

    // latency of the TPG inference
    uint32_t total_cycles = 0;

    for (int c = 0; c < NB_CLASSES; ++c) {
        samples[c].count = 0;
    }

    for (int i = 0; i < NB_TEAMS; ++i)
        team_cycles[i] = 0;

    /* Temporary */
    typeInf actionID = -1;

    for (int seed = 0; seed < NB_SEED; ++seed)
    {
        assign_LE_values(seed);
        
        uint32_t total_cycles = 0;
    
        /* Reset cycle counter */
        CSR_WRITE(CSR_REG_MCYCLE, 0);

        inferenceTPG(&actionID, in1, in2, in3, in4, team_cycles);

        /* Read total cycles */
        CSR_READ(CSR_REG_MCYCLE, &total_cycles);

        /* ===== DEBUG ===== */
        PRINTF("Class ID: %d, Run %d, Total Cycles: %d\n", ids_graph_traversals[seed], total_cycles);
            
        int cls = ids_graph_traversals[seed];
        if (cls < 0 || cls >= NB_CLASSES) {
            PRINTF("Seed %d has invalid class %d\n", seed, cls);
            continue;
        }

        /* Store raw value */
        if (samples[cls].count < NB_SEED) {
            samples[cls].values[samples[cls].count++] = total_cycles;
        }
        else {
            PRINTF("Warning: Class %d has more than %d samples, skipping extra samples.\n", cls, NB_SEED);
        }
    }

    /* ========================= */
    /* ===== COMPUTE STATS ===== */
    /* ========================= */

    //PRINTF("Class,Count,AvgCyclesPerClass,StddevCyclesPerClass\n");
    
    for (int c = 0; c < NB_CLASSES; ++c)
    {
        uint32_t n = samples[c].count;
        if (n == 0) continue;

        // --- mean --- 
        double sum = 0.0;
        for (uint32_t i = 0; i < n; i++) {
            sum += samples[c].values[i];
        }
        double mean = sum / n;

        // --- variance --- 
        double var = 0.0;
        for (uint32_t i = 0; i < n; i++) {
            double diff = samples[c].values[i] - mean;
            var += diff * diff;
        }

        double variance = (n > 1) ? (var / (n - 1)) : 0.0;
        double stddev = sqrt(variance);

        PRINTF("%d,%u,%d,%d\n",
            c,
            n,
            (int)mean,
            (int)stddev
        );
    }

    return 0;
}
